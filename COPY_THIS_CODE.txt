    <script>
        let clients = JSON.parse(localStorage.getItem('clients') || '[]');
        
        const palettes = [
            {id: 1, name: "Spring Bloom", colors: ["#FFB6C1", "#FF69B4", "#98FB98", "#F0FFF0", "#FFE4E1", "#FFF8DC"]},
            {id: 2, name: "Ocean Breeze", colors: ["#87CEEB", "#4682B4", "#5F9EA0", "#B0E0E6", "#E0FFFF", "#F0F8FF"]},
            {id: 3, name: "Autumn Warmth", colors: ["#D2691E", "#CD853F", "#A0522D", "#F4A460", "#DEB887", "#FFEFD5"]},
            {id: 4, name: "Winter Elegance", colors: ["#708090", "#2F4F4F", "#696969", "#C0C0C0", "#F5F5F5", "#FFFFFF"]},
            {id: 5, name: "Berry Delight", colors: ["#8B0000", "#DC143C", "#FF1493", "#FF69B4", "#FFB6C1", "#FFF0F5"]},
            {id: 6, name: "Royal Purple", colors: ["#4B0082", "#663399", "#9370DB", "#BA55D3", "#DDA0DD", "#F8F0FF"]}
        ];

        const suggestions = {
            1: [{name: "Nordstrom", items: ["Floral dresses", "Spring scarves"], url: "https://nordstrom.com"}],
            2: [{name: "J.Crew", items: ["Navy blazers", "Denim shirts"], url: "https://jcrew.com"}],
            3: [{name: "Zara", items: ["Autumn coats", "Warm sweaters"], url: "https://zara.com"}],
            4: [{name: "Theory", items: ["Professional wear", "Gray suits"], url: "https://theory.com"}],
            5: [{name: "Kate Spade", items: ["Bold accessories", "Wine bags"], url: "https://katespade.com"}],
            6: [{name: "Saks", items: ["Luxury items", "Purple jewelry"], url: "https://saks.com"}]
        };

        function showSection(id) {
            document.querySelectorAll('.section').forEach(s => s.classList.remove('active'));
            document.getElementById(id).classList.add('active');
        }

        function showHome() { showSection('home'); }
        function showAdmin() { showSection('admin'); loadPalettes(); loadClients(); }
        function showLogin() { showSection('login'); }

        function loadPalettes() {
            const container = document.getElementById('palettes');
            container.innerHTML = palettes.map(p => 
                `<div class="col-md-4"><div class="palette-card">
                    <h5>${p.name}</h5>
                    <div>${p.colors.map(c => `<div class="color-swatch" style="background:${c}"></div>`).join('')}</div>
                </div></div>`
            ).join('');

            const select = document.getElementById('paletteSelect');
            select.innerHTML = palettes.map(p => `<option value="${p.id}">${p.name}</option>`).join('');
        }

        function loadClients() {
            const table = document.getElementById('clientsTable');
            if (clients.length === 0) {
                table.innerHTML = '<tr><td colspan="5" class="text-center">No clients yet. Add demo data or create new clients.</td></tr>';
                return;
            }
            table.innerHTML = clients.map(c => {
                const palette = palettes.find(p => p.id === c.paletteId);
                return `<tr>
                    <td>${c.name}</td>
                    <td>${c.email}</td>
                    <td><code>${c.code}</code></td>
                    <td>${palette ? palette.name : 'None'}</td>
                    <td><button class="btn btn-sm btn-danger" onclick="deleteClient('${c.id}')">Delete</button></td>
                </tr>`;
            }).join('');
        }

        function addDemoData() {
            if (clients.length === 0) {
                clients = [
                    {id: '1', name: 'Sarah Johnson', email: 'sarah@example.com', code: 'DEMO123', paletteId: 1},
                    {id: '2', name: 'Emily Chen', email: 'emily@example.com', code: 'DEMO456', paletteId: 2}
                ];
                localStorage.setItem('clients', JSON.stringify(clients));
                loadClients();
                alert('Demo data added! Try login with sarah@example.com / DEMO123');
            }
        }

        function showAddClient() {
            new bootstrap.Modal(document.getElementById('addClientModal')).show();
        }

        function addClient() {
            const name = document.getElementById('newClientName').value;
            const email = document.getElementById('newClientEmail').value;
            const paletteId = parseInt(document.getElementById('paletteSelect').value);
            
            if (!name || !email) return alert('Please fill all fields');
            
            const newClient = {
                id: Date.now().toString(),
                name, email, paletteId,
                code: Math.random().toString(36).substr(2, 8).toUpperCase()
            };
            
            clients.push(newClient);
            localStorage.setItem('clients', JSON.stringify(clients));
            loadClients();
            bootstrap.Modal.getInstance(document.getElementById('addClientModal')).hide();
            alert(`Client added! Access code: ${newClient.code}`);
        }

        function deleteClient(id) {
            if (confirm('Delete this client?')) {
                clients = clients.filter(c => c.id !== id);
                localStorage.setItem('clients', JSON.stringify(clients));
                loadClients();
            }
        }

        function clientLogin(e) {
            e.preventDefault();
            const email = document.getElementById('clientEmail').value;
            const code = document.getElementById('clientCode').value;
            const client = clients.find(c => c.email === email && c.code === code);
            
            if (client) {
                document.getElementById('clientName').textContent = client.name;
                showClientPalette(client);
                showSection('clientDash');
            } else {
                alert('Invalid credentials');
            }
        }

        function showClientPalette(client) {
            const palette = palettes.find(p => p.id === client.paletteId);
            if (!palette) return;
            
            const paletteDiv = document.getElementById('clientPalette');
            paletteDiv.innerHTML = `
                <div class="col-12">
                    <div class="palette-card">
                        <h4>${palette.name}</h4>
                        <div class="row">
                            ${palette.colors.map((color, i) => `
                                <div class="col-md-2 text-center">
                                    <div class="client-color-circle" style="background:${color}"></div>
                                    <small>Color ${i+1}<br>${color}</small>
                                </div>
                            `).join('')}
                        </div>
                    </div>
                </div>
            `;

            const suggestionsDiv = document.getElementById('suggestions');
            const shopSuggestions = suggestions[client.paletteId] || [];
            suggestionsDiv.innerHTML = shopSuggestions.map(shop => `
                <div class="col-md-4">
                    <div class="shopping-card">
                        <h5>${shop.name}</h5>
                        ${shop.items.map(item => `<div>${item}</div>`).join('')}
                        <a href="${shop.url}" target="_blank" class="btn btn-primary btn-sm mt-2">Visit Store</a>
                    </div>
                </div>
            `).join('');
        }

        // Initialize
        showHome();
    </script>
